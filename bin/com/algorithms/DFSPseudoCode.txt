	SUB depthFirstTraversal(graph, currentNode, visitedList, stack ) 
		append currentNode to visitedList
		push currentNode on to stack
		FOREACH	(neighbouring node of currentNode) 
			IF neighbouring node NOT in visitedList THEN
				depthFirstTraversal(graph, currentNode, visitedList, stack)
			ENDIF
		ENDFOREACH
		pop currentNode from stack
		RETURN visitedList
	ENDSUB
