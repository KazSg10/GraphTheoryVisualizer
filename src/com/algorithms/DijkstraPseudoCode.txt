SUB Dijkstra(graph, source)
	queue <- []
	traversedNodes <- []
	FOREACH (node v in graph.nodes)
		IF (v is not the source node)
			dist[v] <-INFINITY
			prev[v] <- NULL
		ELSE
			dist[source] <- 0
		ENDIF	
		
		enqueue v in queue
		
	ENDFOREACH	
	
	WHILE (true)
		u <- node in queue with minimum dist[u]
		
		FOREACH (neighbour node v of u in queue that is not in traversedNodes)
			altDist <- dist[u] + graph.edges(u, v)
			IF(altDist < dist[v])
				dist[v] <- altDist
				prev[v] <- u
			ENDIF
		ENDFOREACH
		
		Add u to traversedNodes
		
		IF(all the nodes have been enqueued in queue)
			break from while loop
		ENDIF	
	ENDWHILE	
ENDSUB
