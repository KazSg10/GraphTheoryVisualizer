sub Dijkstra(graph, source)
	Queue = []
	VisitedNodes = []
	FOREACH (node v graph.nodes)
		IF (v is the source)
			dist[source] <- 0
		ELSE
			dist[v] <-INFINITY
			prev[v] <- NULL
		ENDIF	
		
		enqueue v in Queue
		
	ENDFOREACH	
	
	WHILE (true)
		u <- node in Queue with minimum dist[u]
		
		FOREACH (neighbour node v of u in Queue that is not in VisitedNodes)
			altDist <- dist[u] + graph.edges(u, v)
			IF(altDist < dist[v])
				dist[v] <- altDist
				prev[v] <- u
			ENDIF
		ENDFOREACH
		
		Add u to VisitedNodes
		
		IF(all the nodes have been enqueued in Queue)
			break from while loop
		ENDIF	
	ENDWHILE
