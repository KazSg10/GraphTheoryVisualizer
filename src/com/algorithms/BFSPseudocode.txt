	SUB BreadthFirstTraversal(graph, sourceNode)
		visited <- []
		queue <- []
		enqueue sourceNode in queue
		WHILE queue is NOT empty
			currentNode <- dequeue node from queue
			append currentNode to visited
			FOREACH	(neighbouring node of currentNode) 
				IF(visited NOT contains node && queue NOT contains node)
					enqueue node to queue
				ENDIF	
			ENDFOREACH
		ENDWHILE
		RETURN visited;
	ENDSUB
